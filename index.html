Practical 1: Exploratory Data Analysis (EDA) - I
Aim: To calculate descriptive statistics. 


import numpy as np
import pandas as pd
                
data = [10, 20, 30, 40, 50]
                
print("Mean:", np.mean(data))
print("Median:", np.median(data))
print("Mode:", pd.Series(data).mode().tolist())
print("Variance:", np.var(data))
print("Standard Deviation:", np.std(data))
print("Minimum:", np.min(data))
print("Maximum:", np.max(data))

Practical 14: Maximum Likelihood Estimation (MLE) - I
Aim: To estimate mean and variance using MLE. 


import numpy as np
                
data = np.array([50, 60, 70, 80, 90])
                
mean = np.mean(data)
variance = np.var(data, ddof=0)
std = np.sqrt(variance)
                
print("MLE Mean:", mean)
print("MLE Variance:", variance)
print("MLE Standard Deviation:", round(std, 2))

Practical 18: Regularized Regression
Aim: To implement Ridge, Lasso, and Elastic Net regression. 


import numpy as np
from sklearn.linear_model import Ridge, Lasso, ElasticNet

X = np.array([[1], [2], [3], [4], [5]])
y = np.array([2, 4, 5, 4, 5])

models = {
    "Ridge": Ridge(alpha=1.0),
    "Lasso": Lasso(alpha=0.1),
    "Elastic Net": ElasticNet(alpha=0.1, l1_ratio=0.5)
}

for name, model in models.items():
    model.fit(X, y)
    prediction = model.predict([[6]])
    print(name, "Prediction:", round(prediction[0], 2))
