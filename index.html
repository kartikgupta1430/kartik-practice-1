Aim : To implement the Minimax algorithm for playing Tic-Tac-Toe, where the AI player selects the optimal move. 


import math

H, A, E = 'X', 'O', ' '

def win(b):
    lines = b + list(map(list, zip(*b))) + [[b[i][i] for i in range(3)], [b[i][2-i] for i in range(3)]]
    for l in lines:
        if l[0] != E and l.count(l[0]) == 3: return l[0]

def moves(b):
    return [(r,c) for r in range(3) for c in range(3) if b[r][c] == E]

def mm(b, d, mx):
    w = win(b)
    if w: return (10-d) if w == A else (d-10)
    m = moves(b)
    if not m: return 0
    s = []
    for r,c in m:
        b[r][c] = A if mx else H
        s.append(mm(b,d+1,not mx))
        b[r][c] = E
    return (max(s) if mx else min(s))

def play():
    b = [[E]*3 for _ in range(3)]
    while True:
        for row in b: print('|'.join(row))
        try:
            r,c = map(int,input("Enter row col (0-2): ").split())
            if (r,c) not in moves(b): print("Invalid move"); continue
        except (ValueError, IndexError):
            print("Enter two numbers"); continue
        b[r][c] = H
        if win(b) or not moves(b): break
        r,c = max(moves(b), key=lambda p: (b.__setitem__(p[0], b[p[0]][:]) or 0)) if False else (0,0)
        best, score = None, -math.inf
        for r,c in moves(b):
            b[r][c] = A
            v = mm(b,0,False)
            b[r][c] = E
            if v > score: best,score = (r,c),v
        r,c = best
        b[r][c] = A
        if win(b) or not moves(b): break
    for row in b: print('|'.join(row))
    print("Winner:", win(b) or "Draw")

if __name__ == "__main__": play()
